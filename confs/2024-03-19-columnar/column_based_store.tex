\begin{frame}{Consideraciones sobre la normalización}
    \centering
    \large{\textcolor{red}{ La correctitud del modelo relacional depende de su forma normal ($\geq$ 3FN)}}

    \begin{table}[h!]
        \centering
        \begin{tabular}{|p{0.45\textwidth}|p{0.45\textwidth}|}
        \hline
        \multicolumn{1}{|c|}{\cellcolor{green1} \textbf{Ventajas}} &  \multicolumn{1}{|c|}{ \cellcolor{red1} \textbf{Desventajas}} \\ \hline
        Garantiza el cumplimiento de las dependencias funcionales & Aumenta la complejidad del diseño lógico, especialmente para los usuarios finales \\ \hline
        Disminuye la redundancia y por tanto el espacio consumido & Se requiere de JOINs para recuperar la información original \\ \hline
        Aumenta la eficiencia de las operaciones de escritura & Al separar los datos se requieren más operaciones de JOIN para realizar las consultas\\ \hline
        \end{tabular}
    \end{table}

\end{frame}

\begin{frame}{Modelo dimensional}

    Constituye una abstracción intuitiva que facilita la representación y el análisis de un evento o sujeto en términos de hechos y dimensiones.


    \begin{itemize}
        
        \item Tabla de hechos: asociados con métricas que describen el comportamiento del evento: ¿cuánto?, ¿ocurrió?

        \item Tabla de dimensión: permiten caracterizar los hechos y proporcionan el contexto de análisis : ¿qué?, ¿quién?, ¿cuándo?, ¿dónde? 
    \end{itemize}

\end{frame}

\begin{frame}
    \frametitle{Almacenamiento basado en columnas}

    \begin{block}{Principal caracter\'istica}
        Cada bloque de informaci\'on s\'olo contiene datos de una columna.
    \end{block}

    \begin{columns}<2->[t]
        \column{.5\textwidth}
\begin{exampleblock}{Ventajas}
    \begin{itemize}[<+(1)->]
        \item Requiere menos almacenamiento gracias a t\'ecnicas de compresi\'on
        \item Eficiente para consultas que requieren muchos datos pero s\'olo un peque\~no subconjunto de las columnas
        \item C\'omputo en paralelo
    \end{itemize}
\end{exampleblock}

        \column{.5\textwidth}
\begin{alertblock}{Desventajas}
    \begin{itemize}[<+(1)->]
        \item Las operaciones de escritura son costosas
        \item Ineficiente para operar sobre las filas
    \end{itemize}
\end{alertblock}
    \end{columns}

    \note<2>{@NOTE x ejemplo:
        \begin{itemize}
            \item referenciar a valores \'unicos
            \item eliminar redundancia en datos repetidos continuamente (Run-Length Encoding)
            \item eliminaci\'on de nulls llevando m\'ascaras de bits, por ejemplo
        \end{itemize}
    }
\end{frame}
